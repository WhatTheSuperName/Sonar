API

Интеграция
за 5 минут

Один SDK. Все платформы. Полный контроль.

integration.js 
import { Sonar } from '@sonar/sdk';

// Инициализация
const sonar = new Sonar({
  apiKey: 'your_key',
  region: 'eu-west',
});

// Загрузить игру
await sonar.games.upload({
  file: buildFile,
  name: 'My Awesome Game',
  genre: 'action',
});

// Найти матч
const match = await sonar.matchmaking.find({
  mode: 'ranked_4v4',
  maxPing: 100,
});

sonar.on('match_found', (d) => {
  console.log(`Match: ${d.id}`);
});
