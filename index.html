Для разработчиков

Интеграция
за 5 минут

Один SDK. Все платформы. Полный контроль.

integration.js 
import { Sonar } from '@sonar/sdk';

// Инициализация за 1 строку
const sonar = new Sonar({
  apiKey: 'your_api_key',
  region: 'eu-west',
  antiCheat: true,
});

// Поиск матча
const match = await sonar.matchmaking.find({
  mode: 'ranked_4v4',
  maxLatency: 100,
});

// Обработка событий
sonar.on('match_found', (data) => {
  console.log(`Match: ${data.id}`);
  console.log(`Players: ${data.players.length}`);
});

// Аналитика
sonar.analytics.track('level_complete', {
  level: 7,
  time_ms: 42300,
});
